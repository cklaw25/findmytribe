CREATE TABLE profiles (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    role TEXT NOT NULL DEFAULT '',
    company TEXT NOT NULL DEFAULT '',
    bio TEXT NOT NULL DEFAULT '',
    interests JSONB NOT NULL DEFAULT '[]',
    goals TEXT NOT NULL DEFAULT '',
    past_events JSONB NOT NULL DEFAULT '[]',
    mutual_friend_ids JSONB NOT NULL DEFAULT '[]',
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read on profiles" ON profiles FOR SELECT USING (true);
CREATE POLICY "Allow insert on profiles" ON profiles FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow update on profiles" ON profiles FOR UPDATE USING (true) WITH CHECK (true);
